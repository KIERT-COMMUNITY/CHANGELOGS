    COLLATE utf8mb4_unicode_ci;
USE kiert_db;

SET NAMES utf8mb4;
SET time_zone = '+00:00';


-- ============================================================
-- BLOQUE 1: USUARIOS Y AUTENTICACIÓN
-- ============================================================

CREATE TABLE usuarios (
    id                            BIGINT PRIMARY KEY AUTO_INCREMENT,
    nombre_usuario                VARCHAR(20)  NOT NULL UNIQUE,
    email                         VARCHAR(255) NOT NULL UNIQUE,
    password_hash                 VARCHAR(255) NOT NULL,
    foto_perfil_url               VARCHAR(500),
    bio                           VARCHAR(255),
    marco_id                      VARCHAR(50) DEFAULT 'none',

    rol                           VARCHAR(20) NOT NULL DEFAULT 'USER',

    activo                        BOOLEAN NOT NULL DEFAULT FALSE,
    email_verificado              BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_verificacion_email      TIMESTAMP NULL DEFAULT NULL,
    eliminado                     BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_eliminacion             TIMESTAMP NULL DEFAULT NULL,

    fecha_ultimo_cambio_password  TIMESTAMP NULL DEFAULT NULL,
    intentos_login_fallidos       INT NOT NULL DEFAULT 0,
    bloqueado_hasta               TIMESTAMP NULL DEFAULT NULL,

    fecha_creacion                TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion           TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE codigos_verificacion (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    email             VARCHAR(255) NOT NULL,
    codigo            VARCHAR(6) NOT NULL,
    tipo              VARCHAR(30) NOT NULL,
    fecha_creacion    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_expiracion  TIMESTAMP NOT NULL,
    usado             BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_uso         TIMESTAMP NULL DEFAULT NULL,
    ip_solicitante    VARCHAR(45) NULL,
    user_agent        VARCHAR(500) NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE auditoria_usuario (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id        BIGINT NULL,
    email             VARCHAR(255) NULL,
    evento            VARCHAR(60) NOT NULL,
    descripcion       VARCHAR(500) NULL,
    exito             BOOLEAN NOT NULL DEFAULT TRUE,
    ip                VARCHAR(45) NULL,
    user_agent        VARCHAR(500) NULL,
    datos_extra       JSON NULL,
    fecha_evento      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- ✅ BLOQUE 1.5: ANUNCIOS (depende de usuarios)
-- ============================================================
CREATE TABLE anuncios (
    id                    BIGINT PRIMARY KEY AUTO_INCREMENT,
    titulo                VARCHAR(200) NOT NULL,
    texto                 TEXT NOT NULL,

    -- Imagen (Cloudinary)
    imagen_url            VARCHAR(1000) NULL,
    imagen_public_id      VARCHAR(255) NULL,
    imagen_formato        VARCHAR(20) NULL,
    imagen_peso_kb        INT NULL,
    imagen_ancho          INT NULL,
    imagen_alto           INT NULL,

    -- Enlace de destino
    enlace_url            VARCHAR(1000) NULL,

    -- Estado y visibilidad
    activo                BOOLEAN NOT NULL DEFAULT TRUE,
    destacado             BOOLEAN NOT NULL DEFAULT FALSE,

    -- Auditoría: quién lo creó/editó
    creado_por_id         BIGINT NULL,
    actualizado_por_id    BIGINT NULL,

    -- Fechas detalladas
    fecha_creacion        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion   TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_publicacion     TIMESTAMP NULL DEFAULT NULL,
    fecha_expiracion      TIMESTAMP NULL DEFAULT NULL,

    FOREIGN KEY (creado_por_id)      REFERENCES usuarios(id) ON DELETE SET NULL,
    FOREIGN KEY (actualizado_por_id) REFERENCES usuarios(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- BLOQUE 2: POSTS Y CONTENIDO
-- ============================================================

CREATE TABLE posts (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    autor_id             BIGINT NOT NULL,
    titulo               VARCHAR(120) NOT NULL,
    descripcion          TEXT NOT NULL,
    categoria            VARCHAR(20) NOT NULL,
    total_compartidos    BIGINT NOT NULL DEFAULT 0,
    fecha_creacion       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion    TIMESTAMP NULL DEFAULT NULL,
    eliminado            BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE adjuntos (
    id                 BIGINT PRIMARY KEY AUTO_INCREMENT,
    post_id            BIGINT NOT NULL,
    tipo               VARCHAR(20) NOT NULL,
    nombre             VARCHAR(255) NOT NULL,
    url                VARCHAR(1000) NOT NULL,
    peso_kb            INT,
    duracion_segundos  INT NULL,
    ancho              INT NULL,
    alto               INT NULL,
    formato            VARCHAR(20) NULL,
    fecha_creacion     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE comentarios (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    post_id              BIGINT NOT NULL,
    autor_id             BIGINT NOT NULL,
    contenido            TEXT NOT NULL,
    url_imagen           VARCHAR(500) NULL,
    fecha_creacion       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion    TIMESTAMP NULL DEFAULT NULL,
    eliminado            BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (post_id)  REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE respuestas_comentarios (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    comentario_id        BIGINT NOT NULL,
    autor_id             BIGINT NOT NULL,
    contenido            TEXT NOT NULL,
    url_imagen           VARCHAR(500) NULL,
    fecha_creacion       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion    TIMESTAMP NULL DEFAULT NULL,
    eliminado            BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (comentario_id) REFERENCES comentarios(id) ON DELETE CASCADE,
    FOREIGN KEY (autor_id)      REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE reacciones (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id      BIGINT NOT NULL,
    post_id         BIGINT NULL,
    comentario_id   BIGINT NULL,
    tipo            VARCHAR(10) NOT NULL,
    fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id)    REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id)       REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (comentario_id) REFERENCES comentarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_post       (usuario_id, post_id),
    UNIQUE KEY uk_usuario_comentario (usuario_id, comentario_id),
    CONSTRAINT chk_post_o_comentario CHECK (
        (post_id IS NOT NULL AND comentario_id IS NULL) OR
        (post_id IS NULL AND comentario_id IS NOT NULL)
    )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE reacciones_respuestas (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id      BIGINT NOT NULL,
    respuesta_id    BIGINT NOT NULL,
    tipo            VARCHAR(10) NOT NULL,
    fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id)   REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (respuesta_id) REFERENCES respuestas_comentarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_respuesta (usuario_id, respuesta_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE compartidos (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id        BIGINT NOT NULL,
    post_id           BIGINT NOT NULL,
    tipo_compartido   VARCHAR(20) NOT NULL DEFAULT 'INTERNO',
    comentario        VARCHAR(500) NULL,
    fecha_creacion    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id)    REFERENCES posts(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE OR REPLACE VIEW v_estadisticas_generales AS
SELECT
    (SELECT COUNT(*) FROM usuarios WHERE eliminado = FALSE)                    AS total_usuarios,
    (SELECT COUNT(*) FROM usuarios WHERE email_verificado = TRUE)              AS usuarios_verificados,
    (SELECT COUNT(*) FROM usuarios WHERE activo = TRUE AND eliminado = FALSE)  AS usuarios_activos,
    (SELECT COUNT(*) FROM usuarios WHERE rol IN ('CEO','ADMIN','MODERADOR'))   AS total_admins,
    (SELECT COUNT(*) FROM posts WHERE eliminado = FALSE)                       AS total_posts,
    (SELECT COUNT(*) FROM comentarios WHERE eliminado = FALSE)                 AS total_comentarios,
    (SELECT COUNT(*) FROM grupos_chat WHERE activo = TRUE)                     AS total_grupos,
    (SELECT COUNT(*) FROM mensajes WHERE eliminado = FALSE)                    AS total_mensajes,
    (SELECT COUNT(*) FROM documentos WHERE activo = TRUE)                      AS total_documentos,
    (SELECT COUNT(*) FROM recursos_biblioteca WHERE activo = TRUE)             AS total_recursos,
    (SELECT COUNT(*) FROM reportes WHERE estado = 'PENDIENTE')                 AS reportes_pendientes,
    (SELECT COUNT(*) FROM compartidos)                                         AS total_compartidos,
    (SELECT COUNT(*) FROM notificaciones WHERE leida = FALSE)                  AS notificaciones_no_leidas,
    (SELECT COUNT(*) FROM anuncios WHERE activo = TRUE)                        AS total_anuncios,
    NOW()                                                                      AS fecha_consulta;

CREATE OR REPLACE VIEW v_recursos_biblioteca AS
SELECT
    r.id, r.usuario_id, r.es_usuario, r.titulo, r.descripcion, r.url,
    r.categoria, r.subcategoria, r.imagen, r.autor, r.plataforma,
    r.duracion, r.nivel, r.destacado, r.activo, r.tags,
    r.fecha_agregado, r.fecha_actualizacion,
    u.nombre_usuario  AS creador_nombre,
    u.foto_perfil_url AS creador_foto
FROM recursos_biblioteca r
LEFT JOIN usuarios u ON r.usuario_id = u.id
WHERE r.activo = TRUE;

CREATE OR REPLACE VIEW v_reportes_pendientes AS
SELECT
    r.id,
    r.tipo_reporte,
    r.motivo,
    r.descripcion,
    r.estado,
    r.fecha_creacion,
    ur.id              AS reportante_id,
    ur.nombre_usuario  AS reportante_nombre,
    ud.id              AS reportado_id,
    ud.nombre_usuario  AS reportado_nombre,
    p.titulo           AS post_titulo,
    c.contenido        AS comentario_contenido
FROM reportes r
JOIN usuarios ur ON r.usuario_reportante_id = ur.id
LEFT JOIN usuarios ud ON r.usuario_reportado_id = ud.id
LEFT JOIN posts p ON r.post_id = p.id
LEFT JOIN comentarios c ON r.comentario_id = c.id
WHERE r.estado IN ('PENDIENTE', 'REVISANDO')
ORDER BY r.fecha_creacion ASC;

CREATE OR REPLACE VIEW v_admins AS
SELECT
    id,
    nombre_usuario,
    email,
    rol,
    activo,
    email_verificado,
    fecha_creacion
FROM usuarios
WHERE rol IN ('CEO', 'ADMIN', 'MODERADOR')
  AND eliminado = FALSE
ORDER BY
    FIELD(rol, 'CEO', 'ADMIN', 'MODERADOR'),
    fecha_creacion;

CREATE OR REPLACE VIEW v_anuncios_publicos AS
SELECT
    a.id,
    a.titulo,
    a.texto,
    a.imagen_url,
    a.enlace_url,
    a.destacado,
    a.fecha_creacion,
    a.fecha_publicacion,
    a.fecha_expiracion
FROM anuncios a
WHERE a.activo = TRUE
  AND (a.fecha_expiracion IS NULL OR a.fecha_expiracion > NOW())
ORDER BY a.destacado DESC, a.fecha_creacion DESC;

DELIMITER $$

CREATE PROCEDURE sp_estadisticas_usuario(
    IN p_usuario_id BIGINT,
    IN p_desde DATE,
    IN p_hasta DATE
)
BEGIN
    SELECT
        u.id,
        u.nombre_usuario,
        COUNT(DISTINCT p.id)             AS total_posts,
        COUNT(DISTINCT c.id)             AS total_comentarios,
        COUNT(DISTINCT r.id)             AS total_reacciones,
        COUNT(DISTINCT s.id)             AS total_solicitudes_enviadas,
        COUNT(DISTINCT m.id)             AS total_grupos
    FROM usuarios u
    LEFT JOIN posts p ON p.autor_id = u.id
        AND p.fecha_creacion BETWEEN p_desde AND p_hasta
        AND p.eliminado = FALSE
    LEFT JOIN comentarios c ON c.autor_id = u.id
        AND c.fecha_creacion BETWEEN p_desde AND p_hasta
        AND c.eliminado = FALSE
    LEFT JOIN reacciones r ON r.usuario_id = u.id
        AND r.fecha_creacion BETWEEN p_desde AND p_hasta
    LEFT JOIN solicitudes_contacto s ON s.emisor_id = u.id
        AND s.fecha_solicitud BETWEEN p_desde AND p_hasta
    LEFT JOIN miembros_grupo m ON m.usuario_id = u.id
        AND m.estado = 'ACTIVO'
    WHERE u.id = p_usuario_id
    GROUP BY u.id;
END$$

CREATE PROCEDURE sp_top_posts(
    IN p_desde DATE,
    IN p_hasta DATE,
    IN p_limite INT
)
BEGIN
    SELECT
        p.id,
        p.titulo,
        p.categoria,
        u.nombre_usuario  AS autor,
        p.total_compartidos,
        (SELECT COUNT(*) FROM reacciones WHERE post_id = p.id)     AS total_reacciones,
        (SELECT COUNT(*) FROM comentarios WHERE post_id = p.id AND eliminado = FALSE) AS total_comentarios
    FROM posts p
    JOIN usuarios u ON p.autor_id = u.id
    WHERE p.eliminado = FALSE
      AND p.fecha_creacion BETWEEN p_desde AND p_hasta
    ORDER BY p.total_compartidos DESC
    LIMIT p_limite;
END$$
