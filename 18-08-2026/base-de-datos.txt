-- =============================================
-- BASE DE DATOS PARA KIERT - COMPLETA Y ACTUALIZADA
-- =============================================
DROP DATABASE IF EXISTS kiert_db;
CREATE DATABASE IF NOT EXISTS kiert_db;
USE kiert_db;

-- =============================================
-- TABLA: usuarios
-- =============================================
CREATE TABLE usuarios (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    nombre_usuario VARCHAR(20) NOT NULL UNIQUE,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    foto_perfil_url VARCHAR(500),
    bio VARCHAR(255),
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    INDEX idx_email (email),
    INDEX idx_nombre_usuario (nombre_usuario),
    INDEX idx_activo (activo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: password_reset_tokens
-- =============================================
CREATE TABLE password_reset_tokens (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    token VARCHAR(36) NOT NULL UNIQUE,
    usuario_id BIGINT NOT NULL,
    fecha_expiracion TIMESTAMP NOT NULL,
    usado BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_token (token),
    INDEX idx_usuario_id (usuario_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: posts (CON SOFT DELETE)
-- =============================================
CREATE TABLE posts (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    autor_id BIGINT NOT NULL,
    titulo VARCHAR(120) NOT NULL,
    descripcion TEXT NOT NULL,
    categoria VARCHAR(20) NOT NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion TIMESTAMP NULL DEFAULT NULL,
    eliminado BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_autor_id (autor_id),
    INDEX idx_categoria (categoria),
    INDEX idx_fecha_creacion (fecha_creacion DESC),
    INDEX idx_eliminado (eliminado),
    INDEX idx_fecha_eliminacion (fecha_eliminacion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: adjuntos (ACTUALIZADA - SOPORTE PARA VIDEOS Y GIFS)
-- =============================================
CREATE TABLE adjuntos (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    post_id BIGINT NOT NULL,
    tipo VARCHAR(20) NOT NULL,
    nombre VARCHAR(255) NOT NULL,
    url VARCHAR(1000) NOT NULL,
    peso_kb INT,
    duracion_segundos INT NULL,
    ancho INT NULL,
    alto INT NULL,
    formato VARCHAR(20) NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE,
    INDEX idx_post_id (post_id),
    INDEX idx_tipo (tipo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: comentarios (CON IMÁGENES)
-- =============================================
CREATE TABLE comentarios (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    post_id BIGINT NOT NULL,
    autor_id BIGINT NOT NULL,
    contenido TEXT NOT NULL,
    url_imagen VARCHAR(500) NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion TIMESTAMP NULL DEFAULT NULL,
    eliminado BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_post_id (post_id),
    INDEX idx_autor_id (autor_id),
    INDEX idx_fecha_creacion (fecha_creacion ASC),
    INDEX idx_eliminado (eliminado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: respuestas_comentarios (RESPUESTAS A COMENTARIOS CON IMÁGENES)
-- =============================================
CREATE TABLE respuestas_comentarios (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    comentario_id BIGINT NOT NULL,
    autor_id BIGINT NOT NULL,
    contenido TEXT NOT NULL,
    url_imagen VARCHAR(500) NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion TIMESTAMP NULL DEFAULT NULL,
    eliminado BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (comentario_id) REFERENCES comentarios(id) ON DELETE CASCADE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_comentario_id (comentario_id),
    INDEX idx_autor_id (autor_id),
    INDEX idx_fecha_creacion (fecha_creacion ASC),
    INDEX idx_eliminado (eliminado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: reacciones (REACCIONES A POSTS Y COMENTARIOS)
-- =============================================
CREATE TABLE reacciones (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id BIGINT NOT NULL,
    post_id BIGINT NULL,
    comentario_id BIGINT NULL,
    tipo VARCHAR(10) NOT NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (comentario_id) REFERENCES comentarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_post (usuario_id, post_id),
    UNIQUE KEY uk_usuario_comentario (usuario_id, comentario_id),
    INDEX idx_usuario_id (usuario_id),
    INDEX idx_post_id (post_id),
    INDEX idx_comentario_id (comentario_id),
    INDEX idx_tipo (tipo),
    CONSTRAINT chk_post_o_comentario CHECK (
        (post_id IS NOT NULL AND comentario_id IS NULL) OR
        (post_id IS NULL AND comentario_id IS NOT NULL)
    )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: reacciones_respuestas (REACCIONES A RESPUESTAS)
-- =============================================
CREATE TABLE reacciones_respuestas (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id BIGINT NOT NULL,
    respuesta_id BIGINT NOT NULL,
    tipo VARCHAR(10) NOT NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (respuesta_id) REFERENCES respuestas_comentarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_respuesta (usuario_id, respuesta_id),
    INDEX idx_usuario_id (usuario_id),
    INDEX idx_respuesta_id (respuesta_id),
    INDEX idx_tipo (tipo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: mensajes (CHAT - CON SOPORTE PARA ARCHIVOS)
-- =============================================
CREATE TABLE mensajes (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    emisor_id BIGINT NOT NULL,
    receptor_id BIGINT NOT NULL,
    contenido TEXT NOT NULL,
    fecha_envio TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    leido BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_leido TIMESTAMP NULL DEFAULT NULL,
    tipo_mensaje VARCHAR(20) DEFAULT 'TEXTO',
    url_archivo VARCHAR(1000),
    nombre_archivo VARCHAR(255),
    fecha_eliminacion TIMESTAMP NULL DEFAULT NULL,
    eliminado BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (emisor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (receptor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_emisor_receptor (emisor_id, receptor_id),
    INDEX idx_receptor_leido (receptor_id, leido),
    INDEX idx_fecha_envio (fecha_envio DESC),
    INDEX idx_eliminado (eliminado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: solicitudes_contacto (SOLICITUDES DE AMISTAD)
-- =============================================
CREATE TABLE solicitudes_contacto (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    emisor_id BIGINT NOT NULL,
    receptor_id BIGINT NOT NULL,
    estado VARCHAR(10) NOT NULL DEFAULT 'PENDIENTE',
    mensaje VARCHAR(255) NULL,
    fecha_solicitud TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_respuesta TIMESTAMP NULL DEFAULT NULL,
    FOREIGN KEY (emisor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (receptor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_emisor_receptor (emisor_id, receptor_id),
    INDEX idx_estado (estado),
    INDEX idx_fecha_solicitud (fecha_solicitud DESC),
    UNIQUE KEY uk_solicitud_unica (emisor_id, receptor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: personalizacion_usuario (PERSONALIZACIÓN)
-- =============================================
CREATE TABLE IF NOT EXISTS personalizacion_usuario (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id BIGINT NOT NULL UNIQUE,
    tema_id VARCHAR(50) DEFAULT 'default',
    marco_id VARCHAR(50) DEFAULT 'none',
    fondo_id VARCHAR(50) DEFAULT 'default',
    foto_perfil_url VARCHAR(500),
    foto_portada_url VARCHAR(500),
    marco_personalizado_url VARCHAR(500),
    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_usuario_id (usuario_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: compras_usuario (COMPRAS)
-- =============================================
CREATE TABLE IF NOT EXISTS compras_usuario (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id BIGINT NOT NULL,
    tipo VARCHAR(20) NOT NULL,
    item_id VARCHAR(50) NOT NULL,
    precio DECIMAL(10,2) NOT NULL,
    fecha_compra TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_item (usuario_id, tipo, item_id),
    INDEX idx_usuario_id (usuario_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: marcos_personalizados (MARCOS SUBIDOS POR USUARIOS)
-- =============================================
CREATE TABLE IF NOT EXISTS marcos_personalizados (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(255),
    url_imagen VARCHAR(500) NOT NULL,
    tipo VARCHAR(20) DEFAULT 'circulo',
    precio DECIMAL(10,2) DEFAULT 0,
    gratis BOOLEAN DEFAULT TRUE,
    activo BOOLEAN DEFAULT TRUE,
    usuario_id BIGINT,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL,
    INDEX idx_activo (activo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- TABLA: fondos_personalizados (FONDOS SUBIDOS POR USUARIOS)
-- =============================================
CREATE TABLE IF NOT EXISTS fondos_personalizados (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(255),
    url_imagen VARCHAR(500),
    tipo VARCHAR(20) DEFAULT 'imagen',
    gradiente TEXT,
    precio DECIMAL(10,2) DEFAULT 0,
    gratis BOOLEAN DEFAULT TRUE,
    activo BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_activo (activo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- CONSULTAS ÚTILES
-- =============================================

-- 1. Ver todos los usuarios
SELECT * FROM usuarios;

-- 2. Ver todos los posts con sus autores y adjuntos
SELECT 
    p.id,
    p.titulo,
    p.descripcion,
    p.categoria,
    p.fecha_creacion,
    u.nombre_usuario AS autor_nombre,
    u.foto_perfil_url AS autor_foto,
    COUNT(a.id) AS total_adjuntos
FROM posts p
JOIN usuarios u ON p.autor_id = u.id
LEFT JOIN adjuntos a ON p.id = a.post_id
WHERE p.eliminado = FALSE
GROUP BY p.id
ORDER BY p.fecha_creacion DESC;

-- 3. Ver todos los adjuntos (incluyendo videos y GIFs)
SELECT 
    a.id,
    p.titulo AS post_titulo,
    a.tipo,
    a.nombre,
    a.url,
    a.peso_kb,
    a.duracion_segundos,
    a.formato
FROM adjuntos a
JOIN posts p ON a.post_id = p.id
ORDER BY a.fecha_creacion DESC;

-- 4. Ver comentarios con respuestas e imágenes
SELECT 
    c.id AS comentario_id,
    c.contenido AS comentario,
    c.url_imagen AS comentario_imagen,
    u.nombre_usuario AS autor_comentario,
    c.fecha_creacion,
    r.id AS respuesta_id,
    r.contenido AS respuesta,
    r.url_imagen AS respuesta_imagen,
    u2.nombre_usuario AS autor_respuesta,
    r.fecha_creacion AS fecha_respuesta
FROM comentarios c
LEFT JOIN respuestas_comentarios r ON c.id = r.comentario_id AND r.eliminado = FALSE
JOIN usuarios u ON c.autor_id = u.id
LEFT JOIN usuarios u2 ON r.autor_id = u2.id
WHERE c.eliminado = FALSE
ORDER BY c.fecha_creacion ASC, r.fecha_creacion ASC;

-- 5. Ver reacciones a posts
SELECT 
    p.id AS post_id,
    p.titulo,
    r.tipo,
    COUNT(*) AS cantidad
FROM reacciones r
JOIN posts p ON r.post_id = p.id
WHERE r.post_id IS NOT NULL
GROUP BY p.id, p.titulo, r.tipo
ORDER BY p.id, r.tipo;

-- 6. Ver reacciones a comentarios
SELECT 
    c.id AS comentario_id,
    c.contenido,
    r.tipo,
    COUNT(*) AS cantidad
FROM reacciones r
JOIN comentarios c ON r.comentario_id = c.id
WHERE r.comentario_id IS NOT NULL
GROUP BY c.id, c.contenido, r.tipo
ORDER BY c.id, r.tipo;

-- 7. Ver reacciones a respuestas
SELECT 
    rc.id AS respuesta_id,
    rc.contenido,
    rr.tipo,
    COUNT(*) AS cantidad
FROM reacciones_respuestas rr
JOIN respuestas_comentarios rc ON rr.respuesta_id = rc.id
GROUP BY rc.id, rc.contenido, rr.tipo
ORDER BY rc.id, rr.tipo;

-- 8. Ver mensajes no leídos de un usuario
SELECT 
    m.id,
    u.nombre_usuario AS emisor,
    m.contenido,
    m.fecha_envio
FROM mensajes m
JOIN usuarios u ON m.emisor_id = u.id
WHERE m.receptor_id = 1 AND m.leido = FALSE AND m.eliminado = FALSE
ORDER BY m.fecha_envio DESC;

-- 9. Ver personalización de usuarios
SELECT 
    u.nombre_usuario,
    p.tema_id,
    p.marco_id,
    p.fondo_id,
    p.foto_perfil_url
FROM personalizacion_usuario p
JOIN usuarios u ON p.usuario_id = u.id;

-- 10. Ver compras de usuarios
SELECT 
    u.nombre_usuario,
    c.tipo,
    c.item_id,
    c.precio,
    c.fecha_compra
FROM compras_usuario c
JOIN usuarios u ON c.usuario_id = u.id
ORDER BY c.fecha_compra DESC;

-- 11. Ver solicitudes de contacto pendientes
SELECT 
    s.id,
    e.nombre_usuario AS emisor,
    r.nombre_usuario AS receptor,
    s.estado,
    s.mensaje,
    s.fecha_solicitud
FROM solicitudes_contacto s
JOIN usuarios e ON s.emisor_id = e.id
JOIN usuarios r ON s.receptor_id = r.id
WHERE s.estado = 'PENDIENTE'
ORDER BY s.fecha_solicitud DESC;

-- 12. Ver total de comentarios por post
SELECT 
    p.id AS post_id,
    p.titulo,
    COUNT(DISTINCT c.id) AS total_comentarios,
    COUNT(DISTINCT r.id) AS total_respuestas
FROM posts p
LEFT JOIN comentarios c ON p.id = c.post_id AND c.eliminado = FALSE
LEFT JOIN respuestas_comentarios r ON c.id = r.comentario_id AND r.eliminado = FALSE
WHERE p.eliminado = FALSE
GROUP BY p.id, p.titulo
ORDER BY total_comentarios DESC;

-- 13. Ver actividad reciente
SELECT 
    'comentario' AS tipo,
    c.id,
    u.nombre_usuario AS usuario,
    c.contenido,
    c.fecha_creacion AS fecha
FROM comentarios c
JOIN usuarios u ON c.autor_id = u.id
WHERE c.eliminado = FALSE
UNION ALL
SELECT 
    'respuesta' AS tipo,
    r.id,
    u.nombre_usuario AS usuario,
    r.contenido,
    r.fecha_creacion AS fecha
FROM respuestas_comentarios r
JOIN usuarios u ON r.autor_id = u.id
WHERE r.eliminado = FALSE
ORDER BY fecha DESC
LIMIT 20;

-- 14. Ver estadísticas de adjuntos
SELECT 
    tipo,
    COUNT(*) AS cantidad,
    AVG(peso_kb) AS peso_promedio_kb
FROM adjuntos
GROUP BY tipo
ORDER BY cantidad DESC;

-- 15. Ver posts con videos
SELECT 
    p.id,
    p.titulo,
    a.nombre AS video_nombre,
    a.url,
    a.duracion_segundos,
    a.formato
FROM adjuntos a
JOIN posts p ON a.post_id = p.id
WHERE a.tipo IN ('video', 'gif')
ORDER BY a.fecha_creacion DESC;
