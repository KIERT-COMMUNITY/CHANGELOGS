-- ============================================================
-- KIERT DATABASE
-- ============================================================
-- Charset: utf8mb4
-- Engine: InnoDB
-- ============================================================

DROP DATABASE IF EXISTS kiert_db;
CREATE DATABASE kiert_db ;
USE kiert_db;


-- ============================================================
-- 1. USUARIOS
-- ============================================================
CREATE TABLE usuarios (
    id                            BIGINT PRIMARY KEY AUTO_INCREMENT,
    nombre_usuario                VARCHAR(20)  NOT NULL UNIQUE,
    email                         VARCHAR(255) NOT NULL UNIQUE,
    password_hash                 VARCHAR(255) NOT NULL,
    foto_perfil_url               VARCHAR(500),
    bio                           VARCHAR(255),
    marco_id                      VARCHAR(50) DEFAULT 'none',

    -- Estado de la cuenta
    activo                        BOOLEAN NOT NULL DEFAULT FALSE,
    email_verificado              BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_verificacion_email      TIMESTAMP NULL DEFAULT NULL,
    en_linea                      BOOLEAN NOT NULL DEFAULT FALSE,
    ultima_conexion               TIMESTAMP NULL DEFAULT NULL,
    eliminado                     BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_eliminacion             TIMESTAMP NULL DEFAULT NULL,

    -- Auditoría de contraseña
    fecha_ultimo_cambio_password  TIMESTAMP NULL DEFAULT NULL,
    intentos_login_fallidos       INT NOT NULL DEFAULT 0,
    bloqueado_hasta               TIMESTAMP NULL DEFAULT NULL,

    -- Fechas
    fecha_creacion                TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion           TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,

    -- Índices
    INDEX idx_email             (email),
    INDEX idx_nombre_usuario    (nombre_usuario),
    INDEX idx_activo            (activo),
    INDEX idx_en_linea          (en_linea),
    INDEX idx_eliminado         (eliminado),
    INDEX idx_email_verificado  (email_verificado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 2. CODIGOS DE VERIFICACION
-- ============================================================
CREATE TABLE codigos_verificacion (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    email             VARCHAR(255) NOT NULL,
    codigo            VARCHAR(6) NOT NULL,
    tipo              VARCHAR(30) NOT NULL,
    fecha_creacion    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_expiracion  TIMESTAMP NOT NULL,
    usado             BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_uso         TIMESTAMP NULL DEFAULT NULL,
    ip_solicitante    VARCHAR(45) NULL,
    user_agent        VARCHAR(500) NULL,

    INDEX idx_codigo_email   (email),
    INDEX idx_codigo_codigo  (codigo),
    INDEX idx_codigo_tipo    (tipo),
    INDEX idx_codigo_usado   (usado),
    INDEX idx_codigo_expira  (fecha_expiracion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 3. PASSWORD RESET TOKENS (legacy)
-- ============================================================
CREATE TABLE password_reset_tokens (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    token             VARCHAR(36) NOT NULL UNIQUE,
    usuario_id        BIGINT NOT NULL,
    fecha_expiracion  TIMESTAMP NOT NULL,
    usado             BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_uso         TIMESTAMP NULL DEFAULT NULL,
    ip_solicitante    VARCHAR(45) NULL,
    fecha_creacion    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_token       (token),
    INDEX idx_usuario_id  (usuario_id),
    INDEX idx_usado       (usado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 4. AUDITORIA USUARIO
-- ============================================================
CREATE TABLE auditoria_usuario (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id        BIGINT NULL,
    email             VARCHAR(255) NULL,
    evento            VARCHAR(60) NOT NULL,
    descripcion       VARCHAR(500) NULL,
    exito             BOOLEAN NOT NULL DEFAULT TRUE,
    ip                VARCHAR(45) NULL,
    user_agent        VARCHAR(500) NULL,
    datos_extra       JSON NULL,
    fecha_evento      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL,
    INDEX idx_auditoria_usuario  (usuario_id),
    INDEX idx_auditoria_email    (email),
    INDEX idx_auditoria_evento   (evento),
    INDEX idx_auditoria_fecha    (fecha_evento DESC),
    INDEX idx_auditoria_exito    (exito)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 5. POSTS
-- ============================================================
CREATE TABLE posts (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    autor_id             BIGINT NOT NULL,
    titulo               VARCHAR(120) NOT NULL,
    descripcion          TEXT NOT NULL,
    categoria            VARCHAR(20) NOT NULL,
    total_compartidos    BIGINT NOT NULL DEFAULT 0,
    fecha_creacion       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion    TIMESTAMP NULL DEFAULT NULL,
    eliminado            BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_autor_id        (autor_id),
    INDEX idx_categoria       (categoria),
    INDEX idx_fecha_creacion  (fecha_creacion DESC),
    INDEX idx_eliminado       (eliminado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 6. ADJUNTOS
-- ============================================================
CREATE TABLE adjuntos (
    id                 BIGINT PRIMARY KEY AUTO_INCREMENT,
    post_id            BIGINT NOT NULL,
    tipo               VARCHAR(20) NOT NULL,
    nombre             VARCHAR(255) NOT NULL,
    url                VARCHAR(1000) NOT NULL,
    peso_kb            INT,
    duracion_segundos  INT NULL,
    ancho              INT NULL,
    alto               INT NULL,
    formato            VARCHAR(20) NULL,
    fecha_creacion     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE,
    INDEX idx_post_id (post_id),
    INDEX idx_tipo    (tipo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 7. COMENTARIOS
-- ============================================================
CREATE TABLE comentarios (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    post_id              BIGINT NOT NULL,
    autor_id             BIGINT NOT NULL,
    contenido            TEXT NOT NULL,
    url_imagen           VARCHAR(500) NULL,
    fecha_creacion       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion    TIMESTAMP NULL DEFAULT NULL,
    eliminado            BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (post_id)  REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (autor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_post_id        (post_id),
    INDEX idx_autor_id       (autor_id),
    INDEX idx_fecha_creacion (fecha_creacion ASC),
    INDEX idx_eliminado      (eliminado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 8. RESPUESTAS COMENTARIOS
-- ============================================================
CREATE TABLE respuestas_comentarios (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    comentario_id        BIGINT NOT NULL,
    autor_id             BIGINT NOT NULL,
    contenido            TEXT NOT NULL,
    url_imagen           VARCHAR(500) NULL,
    fecha_creacion       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    fecha_eliminacion    TIMESTAMP NULL DEFAULT NULL,
    eliminado            BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (comentario_id) REFERENCES comentarios(id) ON DELETE CASCADE,
    FOREIGN KEY (autor_id)      REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_comentario_id  (comentario_id),
    INDEX idx_autor_id       (autor_id),
    INDEX idx_fecha_creacion (fecha_creacion ASC),
    INDEX idx_eliminado      (eliminado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 9. REACCIONES
-- ============================================================
CREATE TABLE reacciones (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id      BIGINT NOT NULL,
    post_id         BIGINT NULL,
    comentario_id   BIGINT NULL,
    tipo            VARCHAR(10) NOT NULL,
    fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id)    REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id)       REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (comentario_id) REFERENCES comentarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_post       (usuario_id, post_id),
    UNIQUE KEY uk_usuario_comentario (usuario_id, comentario_id),
    INDEX idx_usuario_id    (usuario_id),
    INDEX idx_post_id       (post_id),
    INDEX idx_comentario_id (comentario_id),
    INDEX idx_tipo          (tipo),
    CONSTRAINT chk_post_o_comentario CHECK (
        (post_id IS NOT NULL AND comentario_id IS NULL) OR
        (post_id IS NULL AND comentario_id IS NOT NULL)
    )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 10. REACCIONES RESPUESTAS
-- ============================================================
CREATE TABLE reacciones_respuestas (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id      BIGINT NOT NULL,
    respuesta_id    BIGINT NOT NULL,
    tipo            VARCHAR(10) NOT NULL,
    fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id)   REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (respuesta_id) REFERENCES respuestas_comentarios(id) ON DELETE CASCADE,
    UNIQUE KEY uk_usuario_respuesta (usuario_id, respuesta_id),
    INDEX idx_usuario_id   (usuario_id),
    INDEX idx_respuesta_id (respuesta_id),
    INDEX idx_tipo         (tipo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 11. MENSAJES
-- ============================================================
CREATE TABLE mensajes (
    id                 BIGINT PRIMARY KEY AUTO_INCREMENT,
    emisor_id          BIGINT NOT NULL,
    receptor_id        BIGINT NOT NULL,
    contenido          TEXT NOT NULL,
    fecha_envio        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    leido              BOOLEAN NOT NULL DEFAULT FALSE,

    tipo_mensaje       VARCHAR(20) DEFAULT 'TEXTO',
    url_archivo        VARCHAR(1000),

    fecha_eliminacion  TIMESTAMP NULL DEFAULT NULL,
    eliminado          BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (emisor_id)   REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (receptor_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_emisor_receptor (emisor_id, receptor_id),

    INDEX idx_fecha_envio     (fecha_envio DESC),

-- ============================================================

-- ============================================================
-- 17. NOTIFICACIONES
-- ============================================================
CREATE TABLE notificaciones (
    id                  BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_destino_id  BIGINT NOT NULL,
    usuario_origen_id   BIGINT NULL,
    tipo                VARCHAR(30) NOT NULL,
    mensaje             VARCHAR(500) NOT NULL,
    leida               BOOLEAN NOT NULL DEFAULT FALSE,
    post_id             BIGINT NULL,
    comentario_id       BIGINT NULL,
    respuesta_id        BIGINT NULL,
    grupo_id            BIGINT NULL,
    url                 VARCHAR(500),
    fecha_creacion      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_leida         TIMESTAMP NULL DEFAULT NULL,
    FOREIGN KEY (usuario_destino_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_origen_id)  REFERENCES usuarios(id) ON DELETE SET NULL,
    FOREIGN KEY (post_id)            REFERENCES posts(id) ON DELETE CASCADE,
    FOREIGN KEY (comentario_id)      REFERENCES comentarios(id) ON DELETE CASCADE,
    FOREIGN KEY (respuesta_id)       REFERENCES respuestas_comentarios(id) ON DELETE CASCADE,
    FOREIGN KEY (grupo_id)           REFERENCES grupos_chat(id) ON DELETE CASCADE,
    INDEX idx_usuario_destino (usuario_destino_id),
    INDEX idx_usuario_origen  (usuario_origen_id),
    INDEX idx_tipo            (tipo),
    INDEX idx_leida           (leida),
    INDEX idx_fecha_creacion  (fecha_creacion DESC),
    INDEX idx_post_id         (post_id),
    INDEX idx_comentario_id   (comentario_id),
    INDEX idx_respuesta_id    (respuesta_id),
    INDEX idx_grupo_id        (grupo_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- 23. RECURSOS BIBLIOTECA
-- ============================================================
CREATE TABLE recursos_biblioteca (
    id                   BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id           BIGINT NULL,
    es_usuario           BOOLEAN NOT NULL DEFAULT FALSE,
    titulo               VARCHAR(200) NOT NULL,
    descripcion          VARCHAR(1000) NULL,
    url                  VARCHAR(1000) NOT NULL,
    categoria            VARCHAR(50) NOT NULL,
    subcategoria         VARCHAR(100) NULL,
    imagen               VARCHAR(500) NULL,
    autor                VARCHAR(150) NULL,
    plataforma           VARCHAR(150) NULL,
    duracion             VARCHAR(80) NULL,
    nivel                VARCHAR(30) NULL,
    destacado            BOOLEAN DEFAULT FALSE,
    activo               BOOLEAN DEFAULT TRUE,
    tags                 VARCHAR(1000) NULL,
    fecha_agregado       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion  TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_categoria       (categoria),
    INDEX idx_destacado       (destacado),
    INDEX idx_activo          (activo),
    INDEX idx_usuario_id      (usuario_id),
    INDEX idx_es_usuario      (es_usuario),
    INDEX idx_usuario_fecha   (usuario_id, fecha_agregado DESC),
    INDEX idx_fecha_agregado  (fecha_agregado DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

    r.id, r.usuario_id, r.es_usuario, r.titulo, r.descripcion, r.url,
    r.categoria, r.subcategoria, r.imagen, r.autor, r.plataforma,
    r.duracion, r.nivel, r.destacado, r.activo, r.tags,
    r.fecha_agregado, r.fecha_actualizacion,
    u.nombre_usuario  AS creador_nombre,
    u.foto_perfil_url AS creador_foto
FROM recursos_biblioteca r
LEFT JOIN usuarios u ON r.usuario_id = u.id
WHERE r.activo = TRUE;


-- ============================================================
-- 24. REPORTES
-- ============================================================
CREATE TABLE reportes (
    id                     BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_reportante_id  BIGINT NOT NULL,
    tipo_reporte           VARCHAR(20) NOT NULL,
    motivo                 VARCHAR(50) NOT NULL,
    descripcion            TEXT NOT NULL,
    post_id                BIGINT NULL,
    comentario_id          BIGINT NULL,
    respuesta_id           BIGINT NULL,
    usuario_reportado_id   BIGINT NULL,
    estado                 VARCHAR(20) NOT NULL DEFAULT 'PENDIENTE',
    revisado_por_id        BIGINT NULL,
    fecha_revision         TIMESTAMP NULL DEFAULT NULL,
    nota_moderador         TEXT NULL,
    accion_tomada          VARCHAR(50) NULL,
    fecha_creacion         TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ip_usuario             VARCHAR(45) NULL,
    FOREIGN KEY (usuario_reportante_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id)               REFERENCES posts(id) ON DELETE SET NULL,
    FOREIGN KEY (comentario_id)         REFERENCES comentarios(id) ON DELETE SET NULL,
    FOREIGN KEY (respuesta_id)          REFERENCES respuestas_comentarios(id) ON DELETE SET NULL,
    FOREIGN KEY (usuario_reportado_id)  REFERENCES usuarios(id) ON DELETE SET NULL,
    FOREIGN KEY (revisado_por_id)       REFERENCES usuarios(id) ON DELETE SET NULL,
    INDEX idx_usuario_reportante (usuario_reportante_id),
    INDEX idx_tipo_reporte       (tipo_reporte),
    INDEX idx_estado             (estado),
    INDEX idx_fecha_creacion     (fecha_creacion DESC),
    INDEX idx_post_id            (post_id),
    INDEX idx_usuario_reportado  (usuario_reportado_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 25. HISTORIAL REPORTES
-- ============================================================
CREATE TABLE historial_reportes (
    id               BIGINT PRIMARY KEY AUTO_INCREMENT,
    reporte_id       BIGINT NOT NULL,
    estado_anterior  VARCHAR(20) NULL,
    estado_nuevo     VARCHAR(20) NOT NULL,
    cambiado_por_id  BIGINT NOT NULL,
    nota             TEXT NULL,
    fecha_cambio     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (reporte_id)      REFERENCES reportes(id) ON DELETE CASCADE,
    FOREIGN KEY (cambiado_por_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    INDEX idx_reporte_id    (reporte_id),
    INDEX idx_fecha_cambio  (fecha_cambio DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- 26. COMPARTIDOS
-- ============================================================
CREATE TABLE compartidos (
    id                BIGINT PRIMARY KEY AUTO_INCREMENT,
    usuario_id        BIGINT NOT NULL,
    post_id           BIGINT NOT NULL,
    tipo_compartido   VARCHAR(20) NOT NULL DEFAULT 'INTERNO',
    comentario        VARCHAR(500) NULL,
    fecha_creacion    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id)    REFERENCES posts(id) ON DELETE CASCADE,
    INDEX idx_usuario_id       (usuario_id),
    INDEX idx_post_id          (post_id),
    INDEX idx_tipo_compartido  (tipo_compartido),
    INDEX idx_fecha_creacion   (fecha_creacion DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- CONSULTAS DE VERIFICACION
-- ============================================================

-- 1. Ver usuarios
SELECT 
    id, 
    nombre_usuario, 
    email, 
    activo, 
    email_verificado, 
    en_linea, 
    fecha_creacion 
FROM usuarios;

-- 2. Ver auditoria reciente
SELECT
    id,
    usuario_id,
    email,
    evento,
    descripcion,
    exito,
    ip,
    fecha_evento
FROM auditoria_usuario
ORDER BY fecha_evento DESC
LIMIT 20;

-- 3. Ver codigos de verificacion activos
SELECT
    id,
    email,
    codigo,
    tipo,
    fecha_creacion,
    fecha_expiracion,
    usado
FROM codigos_verificacion
WHERE usado = FALSE AND fecha_expiracion > NOW()
ORDER BY fecha_creacion DESC;

-- 4. Ver recursos biblioteca con creador
SELECT
    r.id                                       AS ID,
    IF(r.es_usuario, 'Usuario', 'Global')      AS Tipo,
    r.titulo                                   AS Titulo,
    r.descripcion                              AS Descripcion,
    r.categoria                                AS Categoria,
    r.autor                                    AS Autor,
    r.destacado                                AS Destacado,
    COALESCE(u.nombre_usuario, 'Sistema')      AS Creador,
    r.fecha_agregado                           AS Fecha
FROM recursos_biblioteca r
LEFT JOIN usuarios u ON r.usuario_id = u.id
WHERE r.activo = TRUE
ORDER BY r.es_usuario DESC, r.fecha_agregado DESC;

-- 5. Ver grupos con conteo de miembros
SELECT
    g.id                                AS ID,
    g.nombre                            AS Grupo,
    g.tipo                              AS Tipo,
    u.nombre_usuario                    AS Creador,
    COUNT(m.id)                         AS MiembrosActivos,
    g.fecha_creacion                    AS Creado
FROM grupos_chat g
JOIN usuarios u ON g.creador_id = u.id
LEFT JOIN miembros_grupo m ON g.id = m.grupo_id AND m.estado = 'ACTIVO'
WHERE g.activo = TRUE
GROUP BY g.id
ORDER BY MiembrosActivos DESC;

-- 6. Estadisticas generales
SELECT
    (SELECT COUNT(*) FROM usuarios WHERE eliminado = FALSE)                  AS TotalUsuarios,
    (SELECT COUNT(*) FROM usuarios WHERE email_verificado = TRUE)            AS UsuariosVerificados,
    (SELECT COUNT(*) FROM posts WHERE eliminado = FALSE)                     AS TotalPosts,
    (SELECT COUNT(*) FROM grupos_chat WHERE activo = TRUE)                   AS TotalGrupos,
    (SELECT COUNT(*) FROM recursos_biblioteca WHERE activo = TRUE)           AS TotalRecursos,
    (SELECT COUNT(*) FROM documentos WHERE activo = TRUE)                    AS TotalDocumentos,
    (SELECT COUNT(*) FROM auditoria_usuario WHERE evento = 'LOGIN')          AS TotalLogins,
    (SELECT COUNT(*) FROM auditoria_usuario WHERE evento = 'CAMBIO_PASSWORD') AS TotalCambiosPassword;

-- 7. Cambios de contrasena de los ultimos 30 dias
SELECT
    a.id                AS ID,
    u.nombre_usuario    AS Usuario,
    a.email             AS Email,
    a.descripcion       AS Descripcion,
    a.exito             AS Exito,
    a.ip                AS IP,
    a.fecha_evento      AS FechaHora
FROM auditoria_usuario a
LEFT JOIN usuarios u ON a.usuario_id = u.id
WHERE a.evento = 'CAMBIO_PASSWORD'
  AND a.fecha_evento >= NOW() - INTERVAL 30 DAY
ORDER BY a.fecha_evento DESC;

-- 8. Auditoria completa de un usuario (por nombre)
SELECT
    a.evento            AS Evento,
    a.descripcion       AS Descripcion,
    a.exito             AS Exito,
    a.ip                AS IP,
    a.user_agent        AS UserAgent,
    a.fecha_evento      AS FechaHora
FROM auditoria_usuario a
LEFT JOIN usuarios u ON a.usuario_id = u.id
WHERE u.nombre_usuario = 'adrian'
ORDER BY a.fecha_evento DESC;

-- 9. Intentos de login fallidos (para detectar ataques)
SELECT
    ip,
    COUNT(*)             AS IntentosFallidos,
    MAX(fecha_evento)    AS UltimoIntento
FROM auditoria_usuario
WHERE evento = 'LOGIN_FALLIDO'
  AND fecha_evento >= NOW() - INTERVAL 24 HOUR
GROUP BY ip
ORDER BY IntentosFallidos DESC;

-- 10. Usuarios que nunca han cambiado su contrasena
SELECT
    id                              AS ID,
    nombre_usuario                  AS Usuario,
    email                           AS Email,
    fecha_creacion                  AS Registro
FROM usuarios
WHERE fecha_ultimo_cambio_password IS NULL
  AND eliminado = FALSE
ORDER BY fecha_creacion DESC;
